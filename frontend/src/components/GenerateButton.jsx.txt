import { useState } from "react";
import api from "../services/api";
import StatsCard from "../components/StatsCard";

export default function GenerateScript() {
  const [topic, setTopic] = useState("");
  const [result, setResult] = useState("");
  const [loading, setLoading] = useState(false);

  const generate = async () => {
    if (!topic) return;

    setLoading(true);

    try {
      const res = await api.post("/ai/generate-script", {
        topic,
        platform: "Instagram Reels",
        duration: "30 seconds",
        tone: "Motivational",
      });

      setResult(res.data.result);
    } catch (err) {
      alert("Error generating script");
      console.log(err);
    }

    setLoading(false);
  };

  return (
    <div className="space-y-8">

      <div className="grid grid-cols-4 gap-5">
        <StatsCard title="Scripts Generated" value="24" />
        <StatsCard title="Credits Left" value="985" />
        <StatsCard title="History" value="18" />
        <StatsCard title="AI Model" value="Gemini" />
      </div>

      <div className="bg-slate-900 rounded-xl p-8">

        <h1 className="text-3xl font-bold text-white">
          AI Script Generator
        </h1>

        <p className="text-gray-400 mt-2">
          Generate viral Instagram Reels, Shorts and TikTok scripts.
        </p>

        <input
          className="w-full mt-6 bg-slate-800 rounded-lg p-4 text-white outline-none"
          placeholder="Enter your topic..."
          value={topic}
          onChange={(e) => setTopic(e.target.value)}
        />

        <button
          onClick={generate}
          className="mt-6 bg-violet-600 hover:bg-violet-700 text-white px-8 py-3 rounded-lg"
        >
          {loading ? "Generating..." : "✨ Generate Script"}
        </button>

      </div>

      {result && (
        <div className="bg-slate-900 rounded-xl p-8">

          <div className="flex justify-between items-center">

            <h2 className="text-2xl font-bold text-white">
              Generated Script
            </h2>

            <button
              onClick={() => navigator.clipboard.writeText(result)}
              className="bg-violet-600 px-5 py-2 rounded-lg"
            >
              Copy
            </button>

          </div>

          <pre className="text-gray-300 whitespace-pre-wrap mt-6">
            {result}
          </pre>

        </div>
      )}

    </div>
  );
}
